Remove an item to the end of DDL
```py
class Node:
    def __init__(self, value):
        self.value = value
        self.next = None
        self.prev = None
        

class DoublyLinkedList:
    def __init__(self, value):
        new_node = Node(value)
        self.head = new_node
        self.tail = new_node
        self.length = 1

        temp = self.head

        new_node = Node(value)
        if self.head is None:
            self.head = new_node
            self.tail = new_node
        else:
            self.tail.next = new_node
            new_node.prev = self.tail
            self.tail = new_node
        self.length += 1
        return True
    
    def pop(self):
        if self.length == 0:
            return None
        
        temp = self.tail
        if self.length == 1:
            self.head = None
            self.tail = None
        else:
            self.tail = self.tail.prev
            self.tail = temp.prev
            self.tail.next = None
            temp.prev = None
        
        self.length -= 1
        return temp

print(my_doubly_linked_list.pop())

    EXPECTED OUTPUT:
    ----------------
    2
    1
    None

"""
```

```py
        if index < 0 or index > self.length:
            return False

            return self.prepend(value)

        after.prev = new_node
```
